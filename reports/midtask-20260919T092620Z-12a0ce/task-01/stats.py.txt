def summarize_readings(values):
    """Return (min, max, mean) of readings, ignoring None.

    Returns (None, None, None) when there is no reading to summarise.
    The caller's sequence is never mutated.
    """
    valid = [value for value in values if value is not None]
    if not valid:
        return (None, None, None)
    return (min(valid), max(valid), sum(valid) / len(valid))
